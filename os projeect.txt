    // Print leading spaces
    for (int i = 0; i < weekday; i++) {
        printf("    ");
    }

    // Print days of the month
    while (timeinfo.tm_mon == month - 1) {
        printf("%4d", timeinfo.tm_mday);
        if (++weekday > 6) {
            weekday = 0;
            printf("\n");
        } else {
            printf(" ");
        }
        timeinfo.tm_mday++;
        mktime(&timeinfo);
    }
}

#include <stdio.h>
#include <stdlib.h>
#include <alsa/asoundlib.h>

#define PCM_DEVICE "default"

// Function to play audio file
void playAudio(const char *filename) {
    int err;
    snd_pcm_t *pcm_handle;
    snd_pcm_hw_params_t *params;
    unsigned int pcm;
    unsigned int tmp;
    int dir;
    snd_pcm_uframes_t frames;

    // Open PCM device
    if ((err = snd_pcm_open(&pcm_handle, PCM_DEVICE, SND_PCM_STREAM_PLAYBACK, 0)) < 0) {
        printf("Error opening PCM device: %s\n", snd_strerror(err));
        return;
    }

    // Allocate parameters object and fill it with default values
    snd_pcm_hw_params_alloca(&params);
    snd_pcm_hw_params_any(pcm_handle, params);

    // Set parameters: Interleaved mode, 16-bit little-endian, stereo
    snd_pcm_hw_params_set_access(pcm_handle, params, SND_PCM_ACCESS_RW_INTERLEAVED);
    snd_pcm_hw_params_set_format(pcm_handle, params, SND_PCM_FORMAT_S16_LE);
    snd_pcm_hw_params_set_channels(pcm_handle, params, 2);

    // Set sample rate to 44100 Hz
    unsigned int sampleRate = 44100;
    snd_pcm_hw_params_set_rate_near(pcm_handle, params, &sampleRate, &dir);

    // Write parameters to the PCM device
    if ((err = snd_pcm_hw_params(pcm_handle, params)) < 0) {
        printf("Error setting PCM parameters: %s\n", snd_strerror(err));
        return;
    }

    // Prepare PCM device
    if ((err = snd_pcm_prepare(pcm_handle)) < 0) {
        printf("Error preparing PCM device: %s\n", snd_strerror(err));
        return;
    }

    // Open audio file
    FILE *file = fopen(filename, "rb");
    if (!file) {
        printf("Error opening audio file: %s\n", filename);
        return;
    }

    // Read and play audio data
    char buffer[4096];
    while (!feof(file)) {
        if (fread(buffer, sizeof(char), sizeof(buffer), file) > 0) {
            if ((err = snd_pcm_writei(pcm_handle, buffer, sizeof(buffer) / sizeof(buffer[0]))) < 0) {
                printf("Error writing PCM data: %s\n", snd_strerror(err));
                break;
            }
        }
    }

    // Close audio file
    fclose(file);

    // Close PCM device
    snd_pcm_close(pcm_handle);
}

int main() {
    const char *filename = "sample.wav"; // Replace with the path to your audio file
    playAudio(filename);
    return 0;
}
